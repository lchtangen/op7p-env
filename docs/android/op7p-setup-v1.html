PHASE 04

Ventoy USB Fix & Boot Recovery

Watch Out

Plug in the Ventoy USB to the PC, then connect the phone to PC via USB. The USB might appear as a different device path. Always run lsblk first to confirm before mounting.

bash — Find & Mount Ventoy
# Find all block devices
lsblk -f

# Look for 64GB exFAT partition
# Usually /dev/sdb2 or /dev/sdc2

# Install exFAT support
apt install -y exfat-fuse exfatprogs

# Create mount point
mkdir -p /mnt/ventoy

# Mount Ventoy data partition (adjust sdX2)
mount -t exfat /dev/sdb2 /mnt/ventoy

# Verify ISOs are there
ls -lah /mnt/ventoy/
ls /mnt/ventoy/*.iso
bash — Fix Ventoy Config
# Check ventoy folder
ls /mnt/ventoy/ventoy/

# Backup broken config first
cp /mnt/ventoy/ventoy/ventoy.json /mnt/ventoy/ventoy/ventoy.json.bak

# Delete broken config
rm /mnt/ventoy/ventoy/ventoy.json

# Verify ISOs still present
ls /mnt/ventoy/*.iso

# Unmount cleanly
umount /mnt/ventoy

echo "Ventoy config reset — ISOs will show on next boot ✅"

python — Hermes AI Tagger Module
# src/ai/hermes_tagger.py
import anthropic
import librosa
import numpy as np
import soundfile as sf
from pathlib import Path
from dataclasses import dataclass
from typing import Optional
import json

@dataclass
class AudioTags:
    bpm: float
    key: str
    scale: str
    genre: list[str]
    mood: list[str]
    instruments: list[str]
    energy: float
    danceability: float
    tags: list[str]

class HermesTagger:
    def __init__(self, api_key: Optional[str] = None):
        self.client = anthropic.Anthropic(api_key=api_key)
        self.model = "claude-opus-4-6"

    def analyze_audio_features(self, filepath: str) -> dict:
        """Extract low-level audio features with librosa."""
        y, sr = librosa.load(filepath, sr=None)
        tempo, _ = librosa.beat.beat_track(y=y, sr=sr)
        chroma = librosa.feature.chroma_cqt(y=y, sr=sr)
        spectral_centroids = librosa.feature.spectral_centroid(y=y, sr=sr)
        rms = librosa.feature.rms(y=y)
        
        keys = ['C','C#','D','D#','E','F','F#','G','G#','A','A#','B']
        key_idx = np.argmax(np.mean(chroma, axis=1))
        
        return {
            "bpm": round(float(tempo), 1),
            "key": keys[key_idx],
            "energy": float(np.mean(rms)),
            "brightness": float(np.mean(spectral_centroids)),
            "duration": float(len(y) / sr)
        }

    def generate_tags(self, filepath: str) -> AudioTags:
        """Use Claude to intelligently tag audio sample."""
        features = self.analyze_audio_features(filepath)
        filename = Path(filepath).name

        prompt = f"""Analyze this audio sample and generate comprehensive tags.
        
        File: {filename}
        Audio Features:
        - BPM: {features['bpm']}
        - Key: {features['key']}
        - Energy level: {features['energy']:.4f}
        - Duration: {features['duration']:.2f}s
        - Brightness: {features['brightness']:.2f}
        
        Return ONLY valid JSON with these exact fields:
        {{
          "bpm": float,
          "key": "string",
          "scale": "major|minor",
          "genre": ["tag1","tag2"],
          "mood": ["tag1","tag2"],
          "instruments": ["tag1","tag2"],
          "energy": float 0-1,
          "danceability": float 0-1,
          "tags": ["descriptive","tags","list"]
        }}"""

        response = self.client.messages.create(
            model=self.model,
            max_tokens=500,
            messages=[{"role": "user", "content": prompt}]
        )

        data = json.loads(response.content[0].text)
        return AudioTags(**data)
python — CLI Entry Point
# src/cli/main.py
import typer
from rich.console import Console
from rich.table import Table
from rich.progress import track
from rich import print as rprint
from pathlib import Path
from ..ai.hermes_tagger import HermesTagger

app = typer.Typer(help="SampleMind AI — Intelligent Audio Sample Manager")
console = Console()

@app.command()
def tag(
    path: str = typer.Argument(..., help="Audio file or folder path"),
    output: str = typer.Option("json", help="Output format: json|table|csv")
):
    """Tag audio samples using Hermes AI."""
    tagger = HermesTagger()
    audio_path = Path(path)

    if audio_path.is_file():
        files = [audio_path]
    else:
        files = list(audio_path.glob("**/*.wav")) + \
                list(audio_path.glob("**/*.mp3")) + \
                list(audio_path.glob("**/*.flac"))

    table = Table(title="SampleMind AI Tags", style="cyan")
    table.add_column("File", style="white")
    table.add_column("BPM", style="green")
    table.add_column("Key", style="yellow")
    table.add_column("Genre", style="magenta")
    table.add_column("Mood", style="blue")
    table.add_column("Energy", style="red")

    for f in track(files, description="Analyzing samples..."):
        tags = tagger.generate_tags(str(f))
        table.add_row(
            f.name,
            str(tags.bpm),
            f"{tags.key} {tags.scale}",
            ", ".join(tags.genre[:2]),
            ", ".join(tags.mood[:2]),
            f"{tags.energy:.2f}"
        )

    console.print(table)

if __name__ == "__main__":
    app()

PHASE 06

Startup Script & Automation

Pro Tip

Save this script in Termux home and run it every time you want Ubuntu. It mounts everything automatically and drops you into your SampleMind project.

bash — Master Startup Script
#!/bin/bash
# ubuntu.sh — Ubuntu Powerhouse Launcher
# Save to: /data/data/com.termux/files/home/ubuntu.sh
# Run: bash ~/ubuntu.sh

UBUNTU_DIR="/data/local/ubuntu"
USER="tangen"

echo "⬡ Launching Ubuntu Powerhouse..."

su -c "
  # Mount filesystems
  mountpoint -q $UBUNTU_DIR/dev     || mount -o bind /dev $UBUNTU_DIR/dev
  mountpoint -q $UBUNTU_DIR/dev/pts || mount -o bind /dev/pts $UBUNTU_DIR/dev/pts
  mountpoint -q $UBUNTU_DIR/proc    || mount -o bind /proc $UBUNTU_DIR/proc
  mountpoint -q $UBUNTU_DIR/sys     || mount -o bind /sys $UBUNTU_DIR/sys

  # Mount DNS
  mount -o bind /etc/resolv.conf $UBUNTU_DIR/etc/resolv.conf 2>/dev/null || true

  echo '✅ Filesystems mounted'

  # Enter chroot as user
  chroot $UBUNTU_DIR /bin/bash -c '
    source /etc/environment 2>/dev/null
    export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
    export HOME=/home/$USER
    export TERM=xterm
    export TMPDIR=/tmp
    export LANG=en_US.UTF-8
    export DEBIAN_FRONTEND=noninteractive

    # Auto-activate SampleMind venv if exists
    if [ -d /home/$USER/projects/samplemind/venv ]; then
      source /home/$USER/projects/samplemind/venv/bin/activate
      cd /home/$USER/projects/samplemind
      echo \"🎵 SampleMind venv activated\"
    fi

    exec su - $USER
  '
"
