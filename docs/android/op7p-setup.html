PHASE 04

Ventoy USB Fix

Watch Out

Plug Ventoy USB into PC. Run lsblk inside Ubuntu chroot to identify the exact device path before mounting — never assume /dev/sdb.

bash — Find & Mount Ventoy
apt install -y exfat-fuse exfatprogs

# Find all block devices
lsblk -f

# Mount Ventoy data partition (adjust sdX2)
mkdir -p /mnt/ventoy
mount -t exfat /dev/sdb2 /mnt/ventoy

# Verify ISOs
ls -lah /mnt/ventoy/
ls /mnt/ventoy/*.iso
bash — Fix Broken Config
# Backup first
cp /mnt/ventoy/ventoy/ventoy.json \
   /mnt/ventoy/ventoy/ventoy.json.bak 2>/dev/null

# Validate JSON (if exists)
python3 -m json.tool /mnt/ventoy/ventoy/ventoy.json

# Delete broken config
rm /mnt/ventoy/ventoy/ventoy.json

# Verify ISOs still there
ls /mnt/ventoy/*.iso

# Unmount cleanly
umount /mnt/ventoy
echo "Ventoy fixed — ISOs will show on next boot"

Hermes AI Tagger Module

python — src/ai/hermes_tagger.py
import anthropic, librosa, numpy as np, json
from pathlib import Path
from dataclasses import dataclass
from typing import Optional

@dataclass
class AudioTags:
    bpm: float
    key: str
    scale: str
    genre: list
    mood: list
    instruments: list
    energy: float
    danceability: float
    tags: list

class HermesTagger:
    def __init__(self, api_key: Optional[str] = None):
        self.client = anthropic.Anthropic(api_key=api_key)
        self.model = "claude-opus-4-6"

    def analyze_audio(self, filepath: str) -> dict:
        y, sr = librosa.load(filepath, sr=None)
        tempo, _ = librosa.beat.beat_track(y=y, sr=sr)
        chroma = librosa.feature.chroma_cqt(y=y, sr=sr)
        rms = librosa.feature.rms(y=y)
        centroid = librosa.feature.spectral_centroid(y=y, sr=sr)
        keys = ['C','C#','D','D#','E','F','F#','G','G#','A','A#','B']
        return {
            "bpm": round(float(tempo), 1),
            "key": keys[int(np.argmax(np.mean(chroma, axis=1)))],
            "energy": float(np.mean(rms)),
            "brightness": float(np.mean(centroid)),
            "duration": float(len(y) / sr)
        }

    def generate_tags(self, filepath: str) -> AudioTags:
        f = self.analyze_audio(filepath)
        prompt = f"""Analyze audio sample and return ONLY valid JSON.
File: {Path(filepath).name}
BPM:{f['bpm']} Key:{f['key']} Energy:{f['energy']:.4f}
Duration:{f['duration']:.2f}s Brightness:{f['brightness']:.2f}
JSON fields: bpm(float),key,scale,genre[],mood[],
instruments[],energy(0-1),danceability(0-1),tags[]"""
        res = self.client.messages.create(
            model=self.model, max_tokens=500,
            messages=[{"role":"user","content":prompt}])
        return AudioTags(**json.loads(res.content[0].text))
python — src/cli/main.py
import typer
from rich.console import Console
from rich.table import Table
from rich.progress import track
from pathlib import Path
from ..ai.hermes_tagger import HermesTagger

app = typer.Typer(help="SampleMind AI — Audio Sample Manager")
console = Console()

@app.command()
def tag(
    path: str = typer.Argument(..., help="File or folder path"),
    output: str = typer.Option("table", help="table|json|csv")
):
    """Tag audio samples with Hermes AI."""
    tagger = HermesTagger()
    p = Path(path)
    files = [p] if p.is_file() else \
        list(p.glob("**/*.wav")) + \
        list(p.glob("**/*.mp3")) + \
        list(p.glob("**/*.flac"))

    table = Table(title="[cyan]SampleMind AI Results[/]", border_style="cyan")
    [table.add_column(c, style=s) for c, s in [
        ("File","white"),("BPM","green"),("Key","yellow"),
        ("Genre","magenta"),("Mood","blue"),("Energy","red")]]

    for f in track(files, description="Analyzing…"):
        t = tagger.generate_tags(str(f))
        table.add_row(f.name, str(t.bpm), f"{t.key} {t.scale}",
            ", ".join(t.genre[:2]), ", ".join(t.mood[:2]), f"{t.energy:.2f}")

    console.print(table)

if __name__ == "__main__":
    app()

PHASE 06

Startup Script & Automation

Pro Tip

Save ubuntu.sh in Termux home. It mounts all filesystems, fixes DNS, activates the venv, and drops you directly into SampleMind project. Run with: ubuntu

bash — ~/ubuntu.sh (Termux)
#!/bin/bash
# Ubuntu Powerhouse Launcher — save to ~/ubuntu.sh

UBUNTU_DIR="/data/local/ubuntu"
USER="tangen"

echo "⬡ Launching Ubuntu Powerhouse..."

su -c "
  mountpoint -q $UBUNTU_DIR/dev     || mount -o bind /dev $UBUNTU_DIR/dev
  mountpoint -q $UBUNTU_DIR/dev/pts || mount -o bind /dev/pts $UBUNTU_DIR/dev/pts
  mountpoint -q $UBUNTU_DIR/proc    || mount -o bind /proc $UBUNTU_DIR/proc
  mountpoint -q $UBUNTU_DIR/sys     || mount -o bind /sys $UBUNTU_DIR/sys
  mount -o bind /etc/resolv.conf $UBUNTU_DIR/etc/resolv.conf 2>/dev/null || true
  echo '✅ Filesystems mounted'

  chroot $UBUNTU_DIR /bin/bash -c '
    source /etc/environment 2>/dev/null
    export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
    export HOME=/home/$USER TERM=xterm TMPDIR=/tmp
    export LANG=en_US.UTF-8 DEBIAN_FRONTEND=noninteractive
    if [ -d /home/$USER/projects/samplemind/venv ]; then
      source /home/$USER/projects/samplemind/venv/bin/activate
      cd /home/$USER/projects/samplemind
      echo \"🎵 SampleMind venv active\"
    fi
    exec su - $USER
  '
"
